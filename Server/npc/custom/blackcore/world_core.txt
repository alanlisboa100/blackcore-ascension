-	script	BlackCoreWorldEvents	-1,{
OnPCLoginEvent:
	dispbottom "Black Core: Ascension — Bem-vindo, " + strcharinfo(0) + ".";
	dispbottom "Moeda do mundo: Núcleos | Região inicial: Nova Aurora";

	// Character creation can now start directly in one of the six Black Core paths.
	// Keep the original Maya gift behavior, but guarantee direct-start characters
	// receive the same one-time starter kit.
	if (!BC_PathGift) {
		.@weapon = 0;
		switch(Class) {
		case Job_Swordman: .@weapon = 1101; break;
		case Job_Mage: .@weapon = 1601; break;
		case Job_Archer: .@weapon = 1701; break;
		case Job_Acolyte: .@weapon = 1501; break;
		case Job_Merchant: .@weapon = 1301; break;
		case Job_Thief: .@weapon = 1201; break;
		}
		if (.@weapon) {
			getitem .@weapon,1;
			getitem Red_Potion,20;
			BC_PathGift = 1;
			dispbottom "Kit inicial recebido: arma do Caminho + 20 Poções Rubras.";
		}
	}
	end;
}

prontera,159,185,4	script	Caio#bc_scout	4_M_JOB_HUNTER,{

		mes "Você já estabilizou o Primeiro Pulso. Agora procure Maya para avançar seu caminho ou Ravi quando estiver pronto para o Abismo.";
		close;
	}
	if (.@state == 0) {
		mes "Os Campos de Nova Aurora estão cheios de criaturas agitadas.";
		mes "Derrube ^FF777710 Geleias Rosa^000000 e ^FFD1665 Coelhos Lunares^000000.";

		setquest 90001;
		mes "Missão aceita: ^5EEBFFPrimeiro Pulso^000000.";
		mes "Volte quando terminar. Tenho Núcleos e suprimentos para você.";
		close;
	}

	if (checkquest(90001,HUNTING) != 2) {
		mes "Ainda sinto atividade nos campos.";
		mes "Complete os objetivos de ^5EEBFFPrimeiro Pulso^000000 e volte.";

	}

	mes "Excelente trabalho. Nova Aurora já respira um pouco melhor.";
	completequest 90001;

	getexp 2500,1200;

	mes "Recompensas:";
	mes "^FFD1661.500 Núcleos^000000";

	mes "Experiência de combate e classe.";

}
