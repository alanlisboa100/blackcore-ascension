prontera,149,185,4	script	Rafa#bc_party	4_M_01,{
	mes "[Rafa — Organizador]";
	mes "Caçar em grupo rende melhor no Black Core. Posso criar uma equipe para você mesmo enquanto o cliente mobile ganha sua tela social definitiva.";
	next;
	if (getcharid(1)) {
		mes "Você já está em uma equipe: ^5EEBFF" + getpartyname(getcharid(1)) + "^000000.";
		close;
	}
	if (select("Criar equipe:Cancelar") == 2) close;
	mes "Digite um nome curto para a equipe.";
	input .@party_name$;
	if (.@party_name$ == "") close;
	.@result = party_create(.@party_name$);
	if (.@result) {
		mes "Equipe ^5EEBFF" + .@party_name$ + "^000000 criada.";
		mes "Convide outros jogadores e aproveite o bônus de EXP compartilhada.";
	} else {
		mes "Não consegui criar essa equipe. Tente outro nome.";
	}
	close;
}
