//============================================================
//= Black Core: Ascension - streamlined class progression
//= Uses native rAthena Job IDs; only progression presentation changes.
//============================================================

prontera,157,185,4	script	Maya#bc_ascension	4_F_KAFRA1,{
	mes "[Maya — Mestre de Ascensão]";
	mes "Eu desperto caminhos de combate ligados ao Núcleo.";
	mes "Seu caminho atual: ^5EEBFF" + callfunc("F_BlackCoreJobName",Class) + "^000000.";
	next;

	// First path: Novice -> one of six core archetypes.
	if (Class == Job_Novice) {
		if (BaseLevel < 10) {
			mes "Volte no ^FFD166nível base 10^000000.";
			mes "Antes disso, aprenda a sobreviver nos Campos de Nova Aurora.";
			close;
		}
		mes "Escolha seu primeiro caminho. A escolha usa as classes nativas do servidor por baixo, mas a identidade é Black Core.";
		next;
		switch(select("Guerreiro:Arcanista:Arqueiro:Devoto:Mercador:Ladino:Cancelar")) {
		case 1: jobchange Job_Swordman; .@weapon = 1101; break;
		case 2: jobchange Job_Mage; .@weapon = 1601; break;
		case 3: jobchange Job_Archer; .@weapon = 1701; break;
		case 4: jobchange Job_Acolyte; .@weapon = 1501; break;
		case 5: jobchange Job_Merchant; .@weapon = 1301; break;
		case 6: jobchange Job_Thief; .@weapon = 1201; break;
		default: close;
		}
		if (!BC_PathGift) {
			getitem .@weapon,1;
			getitem Red_Potion,20;
			BC_PathGift = 1;
		}
		specialeffect2 EF_LEVEL99;
		mes "Seu primeiro caminho foi despertado.";
		mes "Você recebeu uma arma inicial e ^FF777720 Poções Rubras^000000.";
		close;
	}

	// Second path: Job 40+. This intentionally removes the old quest maze for the mobile alpha.
	if (Class == Job_Swordman || Class == Job_Mage || Class == Job_Archer ||
		Class == Job_Acolyte || Class == Job_Merchant || Class == Job_Thief) {
		if (JobLevel < 40) {
			mes "Seu segundo despertar exige ^FFD166nível de classe 40^000000.";
			mes "Faltam " + (40 - JobLevel) + " níveis de classe.";
			close;
		}
		mes "Seu Núcleo já consegue sustentar uma especialização.";
		next;
		switch(Class) {
		case Job_Swordman:
			switch(select("Cavaleiro:Templário:Cancelar")) {
			case 1: jobchange Job_Knight; break;
			case 2: jobchange Job_Crusader; break;
			default: close;
			}
			break;
		case Job_Mage:
			switch(select("Mago de Guerra:Sábio Arcano:Cancelar")) {
			case 1: jobchange Job_Wizard; break;
			case 2: jobchange Job_Sage; break;
			default: close;
			}
			break;
		case Job_Archer:
			.@performer$ = (Sex == SEX_MALE) ? "Bardo" : "Dançarina";
			switch(select("Caçador:" + .@performer$ + ":Cancelar")) {
			case 1: jobchange Job_Hunter; break;
			case 2:
				if (Sex == SEX_MALE) jobchange Job_Bard;
				else jobchange Job_Dancer;
				break;
			default: close;
			}
			break;
		case Job_Acolyte:
			switch(select("Sacerdote:Monge:Cancelar")) {
			case 1: jobchange Job_Priest; break;
			case 2: jobchange Job_Monk; break;
			default: close;
			}
			break;
		case Job_Merchant:
			switch(select("Mestre Ferreiro:Alquimista:Cancelar")) {
			case 1: jobchange Job_Blacksmith; break;
			case 2: jobchange Job_Alchemist; break;
			default: close;
			}
			break;
		case Job_Thief:
			switch(select("Assassino:Renegado:Cancelar")) {
			case 1: jobchange Job_Assassin; break;
			case 2: jobchange Job_Rogue; break;
			default: close;
			}
			break;
		}
		specialeffect2 EF_LEVEL99;
		mes "Ascensão concluída.";
		close;
	}

	// Third path: streamlined alpha progression, no forced rebirth loop.
	if (Class == Job_Knight || Class == Job_Crusader || Class == Job_Wizard || Class == Job_Sage ||
		Class == Job_Hunter || Class == Job_Bard || Class == Job_Dancer || Class == Job_Priest ||
		Class == Job_Monk || Class == Job_Blacksmith || Class == Job_Alchemist ||
		Class == Job_Assassin || Class == Job_Rogue) {
		if (BaseLevel < 99 || JobLevel < 50) {
			mes "A Ascensão Maior exige ^FFD166nível base 99 e classe 50^000000.";
			close;
		}
		mes "Seu Núcleo está pronto para a Ascensão Maior.";
		if (select("Ascender:Agora não") == 2) close;
		switch(Class) {
		case Job_Knight: jobchange Job_Rune_Knight; break;
		case Job_Crusader: jobchange Job_Royal_Guard; break;
		case Job_Wizard: jobchange Job_Warlock; break;
		case Job_Sage: jobchange Job_Sorcerer; break;
		case Job_Hunter: jobchange Job_Ranger; break;
		case Job_Bard: jobchange Job_Minstrel; break;
		case Job_Dancer: jobchange Job_Wanderer; break;
		case Job_Priest: jobchange Job_Arch_Bishop; break;
		case Job_Monk: jobchange Job_Sura; break;
		case Job_Blacksmith: jobchange Job_Mechanic; break;
		case Job_Alchemist: jobchange Job_Genetic; break;
		case Job_Assassin: jobchange Job_Guillotine_Cross; break;
		case Job_Rogue: jobchange Job_Shadow_Chaser; break;
		}
		specialeffect2 EF_LEVEL99;
		announce strcharinfo(0) + " alcançou uma Ascensão Maior em Black Core!", bc_all;
		mes "Ascensão Maior concluída.";
		close;
	}

	mes "Seu caminho atual já está além dos ritos de Maya.";
	mes "As próximas evoluções serão abertas em uma atualização posterior.";
	close;
}
