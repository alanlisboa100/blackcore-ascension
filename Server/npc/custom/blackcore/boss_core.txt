//============================================================
//= Black Core: Ascension - first public boss encounter
//============================================================

prontera,163,185,4	script	Ravi#bc_abyss	4_M_JOB_HUNTER,{
	mes "[Ravi — Sentinela do Abismo]";
	if (BaseLevel < 30) {
		mes "Ainda é cedo para você encarar o que despertou nos Campos de Nova Aurora.";
		mes "Volte no ^FFD166nível base 30^000000.";
		close;
	}

	.@state = isbegin_quest(90002);
	if (.@state == 2) {
		mes "Você já venceu o Eco do Abismo. Outros bosses chegarão em novas regiões.";
		close;
	}
	if (.@state == 0) {
		mes "Uma criatura conhecida como ^A98BFFSenhor do Abismo^000000 começou a rasgar o campo ao norte.";
		mes "Essa é uma ameaça de grupo. Forme uma equipe antes de invocá-lo.";
		next;
		if (select("Aceitar:Agora não") == 2) close;
		setquest 90002;
		mes "Missão aceita: ^5EEBFFEco do Abismo^000000.";
		close;
	}

	if (checkquest(90002,HUNTING) == 2) {
		mes "Você derrubou o Senhor do Abismo.";
		completequest 90002;
		Zeny += 12000;
		getitem White_Potion,10;
		getitem Blue_Potion,5;
		getexp 20000,10000;
		specialeffect2 EF_LEVEL99;
		mes "Recompensas: ^FFD16612.000 Núcleos^000000, suprimentos e experiência.";
		close;
	}

	mes "A ruptura continua ativa.";
	if (mobcount("prt_fild08","BlackCoreAbyss::OnBossKilled") > 0) {
		mes "O ^A98BFFSenhor do Abismo^000000 já está vivo nos Campos de Nova Aurora.";
		close;
	}
	mes "Posso abrir a ruptura e chamar a criatura. Recomendo uma equipe.";
	next;
	if (select("Invocar boss:Cancelar") == 2) close;
	announce "[Black Core] Uma ruptura se abriu nos Campos de Nova Aurora. O Senhor do Abismo despertou!", bc_all;
	monster "prt_fild08",0,0,"Senhor do Abismo",1039,1,"BlackCoreAbyss::OnBossKilled";
	close;
}

- script BlackCoreAbyss -1,{
OnBossKilled:
	announce "[Black Core] " + strcharinfo(0) + " derrotou o Senhor do Abismo!", bc_all;
	Zeny += 3000;
	getitem Red_Potion,20;
	getexp 5000,2500;
	end;
}
